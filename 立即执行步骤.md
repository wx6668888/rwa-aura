#### 1.2 创建测试文件

**创建文件**: `test/StRWA.test.ts`

```typescript
import { expect } from "chai";
import { ethers } from "hardhat";
import { StRWA } from "../typechain-types";

describe("StRWA", function () {
  let stRWA: StRWA;
  let owner: any;
  let stakingContract: any;

  beforeEach(async function () {
    [owner, stakingContract] = await ethers.getSigners();
    
    const StRWAFactory = await ethers.getContractFactory("StRWA");
    stRWA = await StRWAFactory.deploy();
    await stRWA.waitForDeployment();
    
    // 设置 stakingContract
    await stRWA.setStakingContract(stakingContract.address);
  });

  it("Should mint stRWA tokens", async function () {
    const amount = ethers.parseEther("1000");
    
    // 使用 stakingContract 地址调用 mint
    await stRWA.connect(stakingContract).mint(owner.address, amount);
    
    expect(await stRWA.balanceOf(owner.address)).to.equal(amount);
  });

  it("Should burn stRWA tokens", async function () {
    const amount = ethers.parseEther("1000");
    
    // 先 mint
    await stRWA.connect(stakingContract).mint(owner.address, amount);
    
    // 再 burn
    await stRWA.connect(stakingContract).burn(owner.address, amount);
    
    expect(await stRWA.balanceOf(owner.address)).to.equal(0);
  });
});
```

**创建文件**: `test/SwapContract.test.ts`

```typescript
import { expect } from "chai";
import { ethers } from "hardhat";
import { SwapContract, RWAToken, StRWA } from "../typechain-types";

describe("SwapContract", function () {
  let swapContract: SwapContract;
  let rwaToken: RWAToken;
  let stRwaToken: StRWA;
  let owner: any;
  let user: any;

  beforeEach(async function () {
    [owner, user] = await ethers.getSigners();
    
    // 部署 RWA Token
    const RWATokenFactory = await ethers.getContractFactory("RWAToken");
    rwaToken = await RWATokenFactory.deploy(
      "RWA Token",
      "RWA",
      ethers.parseEther("1000000000"), // 10亿
      owner.address, // treasury
      owner.address  // liquidity
    );
    await rwaToken.waitForDeployment();
    
    // 部署 StRWA
    const StRWAFactory = await ethers.getContractFactory("StRWA");
    stRwaToken = await StRWAFactory.deploy();
    await stRwaToken.waitForDeployment();
    
    // 部署 SwapContract
    const SwapContractFactory = await ethers.getContractFactory("SwapContract");
    swapContract = await SwapContractFactory.deploy(
      await rwaToken.getAddress(),
      await stRwaToken.getAddress()
    );
    await swapContract.waitForDeployment();
    
    // 初始化池子
    const initialAmount = ethers.parseEther("10000");
    await rwaToken.approve(await swapContract.getAddress(), initialAmount);
    await stRwaToken.mint(owner.address, initialAmount);
    await stRwaToken.approve(await swapContract.getAddress(), initialAmount);
    
    await swapContract.initializePool(initialAmount, initialAmount);
  });

  it("Should swap stRWA to RWA", async function () {
    const swapAmount = ethers.parseEther("100");
    
    // 给用户 mint stRWA
    await stRwaToken.mint(user.address, swapAmount);
    await stRwaToken.connect(user).approve(await swapContract.getAddress(), swapAmount);
    
    // 执行互换
    await swapContract.connect(user).swapStRWAToRWA(swapAmount);
    
    // 验证用户获得 RWA
    expect(await rwaToken.balanceOf(user.address)).to.equal(swapAmount);
  });
});
```

### 🟡 第2步：更新部署脚本（明天完成）

#### 2.1 创建完整部署脚本

**创建文件**: `scripts/deploy-all.ts`

```typescript
import { ethers } from "hardhat";
import * as dotenv from "dotenv";

dotenv.config();

async function main() {
  const [deployer] = await ethers.getSigners();
  
  console.log("Deploying contracts with account:", deployer.address);
  console.log("Account balance:", (await ethers.provider.getBalance(deployer.address)).toString());
  
  // 1. 部署 RWAToken
  console.log("\n1. Deploying RWAToken...");
  const RWAToken = await ethers.getContractFactory("RWAToken");
  const rwaToken = await RWAToken.deploy(
    "RWA Token",
    "RWA",
    ethers.parseEther("1000000000"), // 10亿
    process.env.TREASURY_ADDRESS || deployer.address,
    process.env.LIQUIDITY_FUND_ADDRESS || deployer.address
  );
  await rwaToken.waitForDeployment();
  const rwaTokenAddress = await rwaToken.getAddress();
  console.log("RWAToken deployed to:", rwaTokenAddress);
  
  // 2. 部署 StRWA
  console.log("\n2. Deploying StRWA...");
  const StRWA = await ethers.getContractFactory("StRWA");
  const stRWA = await StRWA.deploy();
  await stRWA.waitForDeployment();
  const stRWAAddress = await stRWA.getAddress();
  console.log("StRWA deployed to:", stRWAAddress);
  
  // 3. 部署 StakingContract
  console.log("\n3. Deploying StakingContract...");
  const StakingContract = await ethers.getContractFactory("StakingContract");
  const stakingContract = await StakingContract.deploy(
    process.env.USDT_TOKEN_ADDRESS || "0x0000000000000000000000000000000000000000",
    rwaTokenAddress,
    process.env.TREASURY_ADDRESS || deployer.address,
    process.env.BACKEND_ADDRESS || deployer.address
  );
  await stakingContract.waitForDeployment();
  const stakingContractAddress = await stakingContract.getAddress();
  console.log("StakingContract deployed to:", stakingContractAddress);
  
  // 4. 设置 StRWA 的 stakingContract
  console.log("\n4. Setting StRWA stakingContract...");
  await stRWA.setStakingContract(stakingContractAddress);
  console.log("StRWA stakingContract set");
  
  // 5. 设置 StakingContract 的 stRWA token
  console.log("\n5. Setting StakingContract stRWA token...");
  await stakingContract.setStRWAToken(stRWAAddress);
  console.log("StakingContract stRWA token set");
  
  // 6. 部署 SwapContract
  console.log("\n6. Deploying SwapContract...");
  const SwapContract = await ethers.getContractFactory("SwapContract");
  const swapContract = await SwapContract.deploy(
    rwaTokenAddress,
    stRWAAddress
  );
  await swapContract.waitForDeployment();
  const swapContractAddress = await swapContract.getAddress();
  console.log("SwapContract deployed to:", swapContractAddress);
  
  // 7. 初始化 SwapContract 池子（可选）
  if (process.env.INITIAL_LIQUIDITY_RWA && process.env.INITIAL_LIQUIDITY_ST_RWA) {
    console.log("\n7. Initializing SwapContract pool...");
    const rwaAmount = ethers.parseEther(process.env.INITIAL_LIQUIDITY_RWA);
    const stRwaAmount = ethers.parseEther(process.env.INITIAL_LIQUIDITY_ST_RWA);
    
    await rwaToken.approve(swapContractAddress, rwaAmount);
    await stRWA.mint(deployer.address, stRwaAmount);
    await stRWA.approve(swapContractAddress, stRwaAmount);
    
    await swapContract.initializePool(rwaAmount, stRwaAmount);
    console.log("SwapContract pool initialized");
  }
  
  // 输出部署结果
  console.log("\n=== Deployment Summary ===");
  console.log("RWAToken:", rwaTokenAddress);
  console.log("StRWA:", stRWAAddress);
  console.log("StakingContract:", stakingContractAddress);
  console.log("SwapContract:", swapContractAddress);
  console.log("\n=== Next Steps ===");
  console.log("1. Update .env file with contract addresses");
  console.log("2. Update frontend contract addresses");
  console.log("3. Test on testnet");
}

main()
  .then(() => process.exit(0))
  .catch((error) => {
    console.error(error);
    process.exit(1);
  });
```
