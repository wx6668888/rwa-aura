## 🚨 问题1: 小号刷单防护

### 1.1 攻击场景

**攻击方式**：
```
用户A创建100个小号
├─ 每个小号质押 10 USDT
├─ 小号1推荐小号2，小号2推荐小号3...
└─ 形成推荐链，获得级差奖励

结果：
├─ 投入：100 × 10 = 1,000 USDT
├─ 获得奖励：可能超过 1,000 USDT
└─ 激励池被掏空
```

**问题根源**：
- 没有"有效用户"判定
- 没有最低质押要求
- 没有推荐质量考核

### 1.2 防护方案：多重有效用户判定

```solidity
            users[msg.sender].referrer = referrer;
            emit ReferralBound(msg.sender, referrer, block.timestamp);
```

```solidity
// StakingContract.sol - 推荐质量考核
struct ReferralQuality {
    uint256 directReferrals;        // 直推人数
    uint256 validDirectReferrals;   // 有效直推人数（≥100 USDT）
    uint256 totalTeamStaked;        // 团队总质押额
    uint256 activeTeamMembers;      // 活跃团队成员数
    uint256 qualityScore;           // 质量分数
}

mapping(address => ReferralQuality) public referralQuality;

uint256 public constant MIN_QUALITY_SCORE = 50;  // 最低质量分数

/**
 * @dev 计算推荐质量分数
 */
function calculateQualityScore(address user) public view returns (uint256) {
    ReferralQuality storage quality = referralQuality[user];
    
    if (quality.validDirectReferrals == 0) {
        return 0;
    }
    
    // 质量分数 = 有效直推率 × 50 + 团队平均质押 × 30 + 活跃率 × 20
    uint256 validRate = (quality.validDirectReferrals * 100) / quality.directReferrals;
    uint256 avgStake = quality.totalTeamStaked / quality.validDirectReferrals;
    uint256 activeRate = (quality.activeTeamMembers * 100) / quality.validDirectReferrals;
    
    uint256 score = (validRate * 50 / 100) + 
                    (avgStake / 1000 * 30 / 100) + 
                    (activeRate * 20 / 100);
    
    return score;
}

/**
 * @dev 更新推荐质量
 */
function updateReferralQuality(address referrer, uint256 stakeAmount) internal {
    ReferralQuality storage quality = referralQuality[referrer];
    
    quality.directReferrals++;
    quality.totalTeamStaked += stakeAmount;
    
    // 如果质押金额 ≥ 100 USDT，计入有效直推
    if (stakeAmount >= MIN_VALID_STAKE) {
        quality.validDirectReferrals++;
    }
    
    // 更新质量分数
    quality.qualityScore = calculateQualityScore(referrer);
}

/**
 * @dev 检查推荐人质量
 */
function checkReferrerQuality(address referrer) public view returns (bool) {
    ReferralQuality storage quality = referralQuality[referrer];
    
    // 必须满足最低质量分数
    return quality.qualityScore >= MIN_QUALITY_SCORE;
}
```

#### 方案D：IP/设备指纹检测（后端）

```typescript
// backend/src/services/AntiFraudService.ts
import { query } from '../config/database.config';
import logger from '../utils/logger';

export class AntiFraudService {
    /**
     * 检测小号刷单
     */
    async detectSybilAttack(
        userAddress: string,
        referrerAddress: string,
        stakeAmount: string
    ): Promise<{
        isSuspicious: boolean;
        reason: string;
        riskScore: number;
    }> {
        try {
            let riskScore = 0;
            const reasons: string[] = [];
            
            // 1. 检查最低质押要求
            const minStake = 100 * 10**18;  // 100 USDT
            if (parseInt(stakeAmount) < minStake) {
                return {
                    isSuspicious: true,
                    reason: 'Stake amount below minimum (100 USDT)',
                    riskScore: 100
                };
            }
            
            // 2. 检查推荐人是否为有效用户
            const referrerInfo = await this.getUserInfo(referrerAddress);
            if (!referrerInfo.is_valid_user) {
                return {
                    isSuspicious: true,
                    reason: 'Referrer is not a valid user',
                    riskScore: 100
                };
            }
            
            // 3. 检查推荐链质量
            const referralChain = await this.getReferralChain(referrerAddress);
            const validReferrals = referralChain.filter(r => r.is_valid_user);
            const validRate = validReferrals.length / referralChain.length;
            
            if (validRate < 0.5) {
                riskScore += 30;
                reasons.push(`Low valid referral rate: ${validRate * 100}%`);
            }
            
            // 4. 检查短时间内大量推荐
            const recentReferrals = await this.getRecentReferrals(referrerAddress, 24 * 60 * 60);
            if (recentReferrals.length > 10) {
                riskScore += 40;
                reasons.push(`Too many referrals in 24h: ${recentReferrals.length}`);
            }
            
            // 5. 检查推荐链深度（防止刷单链）
            if (referralChain.length > 20) {
                riskScore += 20;
                reasons.push(`Referral chain too deep: ${referralChain.length}`);
            }
            
            // 6. 检查平均质押金额（防止小号刷单）
            const avgStake = await this.getAverageStake(referralChain);
            if (avgStake < minStake * 2) {
                riskScore += 30;
                reasons.push(`Low average stake: ${avgStake}`);
            }
            
            // 7. 检查IP/设备指纹（如果可用）
            const ipAddress = await this.getUserIP(userAddress);
            const deviceFingerprint = await this.getDeviceFingerprint(userAddress);
            
            const sameIPUsers = await this.getUsersByIP(ipAddress);
            if (sameIPUsers.length > 3) {
                riskScore += 50;
                reasons.push(`Multiple users from same IP: ${sameIPUsers.length}`);
            }
            
            const sameDeviceUsers = await this.getUsersByDevice(deviceFingerprint);
            if (sameDeviceUsers.length > 1) {
                riskScore += 50;
                reasons.push(`Multiple users from same device`);
            }
            
            return {
                isSuspicious: riskScore >= 50,
                reason: reasons.join('; '),
                riskScore
            };
            
        } catch (error) {
            logger.error('Failed to detect sybil attack:', error);
            return {
                isSuspicious: false,
                reason: 'Detection failed',
                riskScore: 0
            };
        }
    }
    
    /**
     * 获取推荐链
     */
    private async getReferralChain(address: string): Promise<any[]> {
        const chain: any[] = [];
        let current = address;
        
        for (let i = 0; i < 20; i++) {  // 最多20代
            const user = await this.getUserInfo(current);
            if (!user.referrer || user.referrer === '0x0000000000000000000000000000000000000000') {
                break;
            }
            chain.push(user);
            current = user.referrer;
        }
        
        return chain;
    }
    
    /**
     * 获取用户信息
     */
    private async getUserInfo(address: string): Promise<any> {
        const result = await query(
            `SELECT * FROM users WHERE address = ?`,
            [address]
        );
        return result[0] || null;
    }
    
    /**
     * 获取最近推荐
     */
    private async getRecentReferrals(referrer: string, timeWindow: number): Promise<any[]> {
        const timestamp = Date.now() - timeWindow * 1000;
        return await query(
            `SELECT * FROM users 
             WHERE referrer = ? AND created_at > ?`,
            [referrer, new Date(timestamp)]
        );
    }
}
```

## 🚨 问题2: 3倍封顶后复投激励

### 2.1 问题分析

**问题场景**：
```
用户达到3倍封顶后：
├─ 不再获得动态收益
├─ 如果RWA币价下跌
├─ 用户可能选择退出
└─ 不再复投
```

**问题根源**：
- RWA币价不受控
- 流动性不足
- 没有复投激励

### 2.2 防护方案：币价稳定 + 复投激励

```solidity
// StakingContract.sol - 复投激励
struct RestakeBonus {
    uint256 bonusRate;      // 复投奖励率
    uint256 unlockTime;     // 解锁时间
    bool claimed;           // 是否已领取
}

mapping(address => RestakeBonus[]) public restakeBonuses;

uint256 public constant RESTAKE_BONUS_RATE = 20;  // 复投奖励20%
uint256 public constant RESTAKE_LOCK_PERIOD = 90 days;  // 锁仓90天

/**
 * @dev 复投（达到封顶后）
 */
function restakeAfterCap(uint256 rwaAmount) external nonReentrant whenNotPaused {
    UserInfo storage user = users[msg.sender];
    
    // 检查是否达到封顶
    require(
        user.totalDynamicRewardsReceived >= user.totalStaked * 3,
        "Not reached cap yet"
    );
    
    // 检查RWA余额
    require(
        rwaToken.balanceOf(msg.sender) >= rwaAmount,
        "Insufficient RWA balance"
    );
    
    // 转换RWA为USDT（按当前价格）
    uint256 usdtAmount = convertRWAToUSDT(rwaAmount);
    
    // 复投奖励：20%
    uint256 bonus = usdtAmount * RESTAKE_BONUS_RATE / 100;
    uint256 totalAmount = usdtAmount + bonus;
    
    // 记录复投奖励（90天后解锁）
    restakeBonuses[msg.sender].push(RestakeBonus({
        bonusRate: RESTAKE_BONUS_RATE,
        unlockTime: block.timestamp + RESTAKE_LOCK_PERIOD,
        claimed: false
    }));
    
    // 更新质押额（提升封顶）
    uint256 internalAmount = totalAmount * PRECISION_MULTIPLIER;
    user.totalStaked += internalAmount;
    
    // 重新计算封顶
    uint256 newCap = user.totalStaked * 3;
    user.totalDynamicRewardsReceived = 0;  // 重置累计收益
    
    // 转账RWA到合约
    rwaToken.safeTransferFrom(msg.sender, address(this), rwaAmount);
    
    // 销毁RWA（通缩）
    rwaToken.burn(rwaAmount);
    
    emit RestakeAfterCap(msg.sender, rwaAmount, bonus, block.timestamp);
}

/**
 * @dev 领取复投奖励
 */
function claimRestakeBonus(uint256 bonusId) external nonReentrant {
    RestakeBonus storage bonus = restakeBonuses[msg.sender][bonusId];
    
    require(!bonus.claimed, "Bonus already claimed");
    require(block.timestamp >= bonus.unlockTime, "Bonus not unlocked yet");
    
    // 计算奖励金额
    uint256 bonusAmount = calculateBonusAmount(msg.sender, bonusId);
    
    // 标记为已领取
    bonus.claimed = true;
    
    // 转账奖励
    usdtToken.safeTransfer(msg.sender, bonusAmount);
    
    emit RestakeBonusClaimed(msg.sender, bonusId, bonusAmount, block.timestamp);
}
```

### 3.2 防护方案：多重熔断机制

#### 方案A：资金池健康度熔断（基础）

```solidity
// StakingContract.sol - 资金池健康度熔断
uint256 public constant CRITICAL_HEALTH_THRESHOLD = 20;  // 临界健康度20%
uint256 public constant EMERGENCY_HEALTH_THRESHOLD = 10;  // 紧急健康度10%

bool public circuitBreakerActive = false;
uint256 public circuitBreakerStartTime;
uint256 public constant CIRCUIT_BREAKER_DURATION = 7 days;  // 熔断持续7天

/**
 * @dev 检查资金池健康度
 */
function checkPoolHealth() public view returns (uint256) {
    uint256 totalStaked = totalStaked;
    uint256 contractBalance = usdtToken.balanceOf(address(this));
    
    // 计算应支付收益

    return health;

function releaseCircuitBreaker() external onlyOwner {
    require(circuitBreakerActive, "Circuit breaker not active");
    require(
        block.timestamp >= circuitBreakerStartTime + CIRCUIT_BREAKER_DURATION,
        "Circuit breaker duration not ended"
    );
    
    uint256 health = checkPoolHealth();
    require(health >= CRITICAL_HEALTH_THRESHOLD, "Pool health still critical");
```

```solidity
contract EmergencyPause is Ownable {

        pausedContracts[contractAddress] = true;

        emit EmergencyPaused(contractAddress, block.timestamp);
    }
    
    /**
     * @dev 解除紧急暂停（仅Owner）
     */
    function emergencyUnpause(address contractAddress) external onlyOwner {
        pausedContracts[contractAddress] = false;
        
        // 检查是否所有合约都已解除暂停
        bool allUnpaused = true;
        // ... 检查逻辑 ...
        
        if (allUnpaused) {
            emergencyPauseActive = false;
        }
        
        emit EmergencyUnpaused(contractAddress, block.timestamp);
    }

     * @dev 检查合约是否暂停
```
