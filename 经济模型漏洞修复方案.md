# 经济模型漏洞修复方案

## 📋 漏洞概述

本文档针对发现的4个关键经济模型漏洞提供详细的修复方案。

**漏洞严重性评级**：
- 🔴 **严重**：动态税率洗白攻击、SwapContract 单向挤兑
- 🟡 **中等**：烧伤机制过于宽松
- 🟢 **低**：合约权限控制（预防性）

## 🚨 漏洞1: 动态税率的"首次质押时间"判定漏洞

### 漏洞描述

**攻击场景**：
```
Day 1: 用户质押 1 USDT → 激活 firstStakeTime
Day 180: 用户质押 1,000,000 USDT
Day 181: 用户卖出 1,000,000 USDT → 仅需支付 10% 税（应该支付 50%）
```

**问题根源**：
- 税率仅基于 `firstStakeTime`（首次质押时间）
- 未考虑资金加权平均持有期
- 未考虑当前质押额度

### 修复方案：资金加权平均持有期

#### 方案A：加权平均持有期（推荐）

```solidity
// StakingContract.sol - 新增结构体
struct StakeRecord {
    uint256 amount;      // 质押金额
    uint256 timestamp;   // 质押时间
    uint256 lockPeriod;  // 锁仓期限（天数）
}

struct UserInfo {
    // ... 现有字段 ...
    StakeRecord[] stakeHistory;  // 质押历史记录
    uint256 totalStaked;         // 总质押额
    uint256 weightedAverageTime; // 加权平均持有时间
}

// 计算加权平均持有期
function calculateWeightedAverageTime(address user) public view returns (uint256) {
    UserInfo storage userInfo = users[user];
    
    if (userInfo.stakeHistory.length == 0) {
        return 0;
    }
    
    uint256 totalWeightedTime = 0;
    uint256 totalAmount = 0;
    
    for (uint256 i = 0; i < userInfo.stakeHistory.length; i++) {
        StakeRecord memory record = userInfo.stakeHistory[i];
        uint256 timeWeight = block.timestamp - record.timestamp;
        uint256 weight = record.amount;
        
        totalWeightedTime += timeWeight * weight;
        totalAmount += weight;
    }
    
    if (totalAmount == 0) {
        return 0;
    }
    
    return totalWeightedTime / totalAmount;  // 加权平均时间（秒）
}

// 修改 stake 函数，记录质押历史
function stake(uint256 amount, address referrer) external nonReentrant whenNotPaused {

        lockPeriod: 0  // 根据选择的锁仓期限设置
    }));
    
    // 更新加权平均持有时间
```

```solidity
// RWAToken.sol - 基于当前质押额度的税率
function calculateSellTaxRate(address user, uint256 sellAmount) internal view returns (uint256) {
    if (stakingContract == address(0)) {
        return SELL_TAX_RATE;  // 20%
    }
    
    // 获取用户当前质押信息
    (bool success, bytes memory data) = stakingContract.staticcall(
        abi.encodeWithSignature("getUserStakeInfo(address)", user)
    );
    
    if (!success || data.length == 0) {
        return SELL_TAX_RATE;  // 默认 20%
    }
    
    (uint256 totalStaked, uint256 weightedAverageTime) = abi.decode(data, (uint256, uint256));
    
    if (totalStaked == 0) {
        return SELL_TAX_RATE;  // 未质押，使用默认税率
    }
    
    // 计算加权平均持有天数
    uint256 weightedHoldingDays = weightedAverageTime / 1 days;
    
    // 基础税率（基于加权平均持有期）
    uint256 baseTaxRate;
    if (weightedHoldingDays < 30) {
        baseTaxRate = 50;
    } else if (weightedHoldingDays < 90) {
        baseTaxRate = 30;
    } else if (weightedHoldingDays < 180) {
        baseTaxRate = 20;
    } else {
        baseTaxRate = 10;
    }
    
    // ========== 新增：基于卖出金额与质押额度的比例调整 ==========
    // 如果卖出金额 > 当前质押额度的 50%，增加税率
    uint256 sellRatio = (sellAmount * 100) / totalStaked;
    
    if (sellRatio > 50) {
        // 卖出金额超过质押额度的50%，税率增加
        uint256 additionalTax = (sellRatio - 50) * 2;  // 每1%增加2%税率
        baseTaxRate = baseTaxRate + additionalTax;
        
        // 最高税率限制为 80%
        if (baseTaxRate > 80) {
            baseTaxRate = 80;
        }
    }
    
    return baseTaxRate;
}
```

## 🚨 漏洞2: SwapContract 单向挤兑风险

### 漏洞描述

**攻击场景**：
```
项目后期，所有人想把 stRWA 换成 RWA
→ RWA 池子被换光
→ SwapContract 瘫痪
→ 用户无法退出
```

**问题根源**：
- 固定比例（1:1 或 1:1.1）无法应对恐慌
- 没有滑点保护
- 没有互换额度限制

### 修复方案：AMM 曲线 + 动态限制

#### 方案A：Uniswap 风格的 x * y = k 曲线（推荐）

```solidity
// SwapContract.sol - 修改为 AMM 曲线
contract SwapContract is Ownable, ReentrancyGuard {
    using SafeERC20 for IERC20;
    
    IERC20 public rwaToken;
    IERC20 public stRwaToken;
    
    // 互换池余额
    uint256 public rwaPoolBalance;
    uint256 public stRwaPoolBalance;
    
    // 常量乘积 k = x * y
    uint256 public constantK;
    
    // 互换参数
    uint256 public constant MIN_LIQUIDITY = 1000 * 10**18;  // 最小流动性
    uint256 public swapFeeRate = 3;  // 0.3% 手续费（可调整）
    
    // 事件
    event SwapStRWAToRWA(address indexed user, uint256 stRwaAmount, uint256 rwaAmount, uint256 fee);
    event SwapRWAToStRWA(address indexed user, uint256 rwaAmount, uint256 stRwaAmount, uint256 fee);
    event PoolUpdated(uint256 rwaBalance, uint256 stRwaBalance, uint256 k);
    
    constructor(address _rwaToken, address _stRwaToken) Ownable(msg.sender) {
        require(_rwaToken != address(0), "Invalid RWA token address");
        require(_stRwaToken != address(0), "Invalid stRWA token address");
        
        rwaToken = IERC20(_rwaToken);
        stRwaToken = IERC20(_stRwaToken);
    }
    
    /**
     * @dev 初始化互换池（项目方提供初始流动性）
     */
    function initializePool(uint256 rwaAmount, uint256 stRwaAmount) external onlyOwner {
        require(rwaPoolBalance == 0 && stRwaPoolBalance == 0, "Pool already initialized");
        require(rwaAmount >= MIN_LIQUIDITY && stRwaAmount >= MIN_LIQUIDITY, "Insufficient initial liquidity");
        
        rwaToken.safeTransferFrom(msg.sender, address(this), rwaAmount);
        stRwaToken.safeTransferFrom(msg.sender, address(this), stRwaAmount);
        
        rwaPoolBalance = rwaAmount;
        stRwaPoolBalance = stRwaAmount;
        constantK = rwaAmount * stRwaAmount;  // k = x * y
        
        emit PoolUpdated(rwaPoolBalance, stRwaPoolBalance, constantK);
    }
    
    /**
     * @dev stRWA → RWA（使用 AMM 曲线）
     * 
     * 公式：x * y = k
     * 输入：Δx (stRWA)
     * 输出：Δy (RWA)
     * 
     * 计算：k = x * y = (x + Δx) * (y - Δy)
     * 解：Δy = (y * Δx) / (x + Δx)
     */
    function swapStRWAToRWA(uint256 stRwaAmount) external nonReentrant returns (uint256) {
        require(stRwaAmount > 0, "Amount must be greater than zero");
        require(rwaPoolBalance > 0 && stRwaPoolBalance > 0, "Pool not initialized");
        
        // 计算手续费
        uint256 fee = (stRwaAmount * swapFeeRate) / 1000;
        uint256 stRwaAmountAfterFee = stRwaAmount - fee;
        
        // 计算输出金额（AMM 公式）
        // Δy = (y * Δx) / (x + Δx)
        uint256 rwaAmount = (rwaPoolBalance * stRwaAmountAfterFee) / (stRwaPoolBalance + stRwaAmountAfterFee);
        
        // 检查最小输出（防止滑点过大）
        require(rwaAmount > 0, "Insufficient output");
        require(rwaPoolBalance >= rwaAmount, "Insufficient RWA pool");
        
        // 检查池子健康度（防止挤兑）
        uint256 newRwaBalance = rwaPoolBalance - rwaAmount;
        uint256 newStRwaBalance = stRwaPoolBalance + stRwaAmountAfterFee;
        
        // 验证 k 值（允许小幅误差）
        uint256 newK = newRwaBalance * newStRwaBalance;
        require(newK >= constantK * 99 / 100, "Pool imbalance detected");  // 允许1%误差
        
        // 转账
        stRwaToken.safeTransferFrom(msg.sender, address(this), stRwaAmount);
        rwaToken.safeTransfer(msg.sender, rwaAmount);
        
        // 更新池子
        rwaPoolBalance = newRwaBalance;
        stRwaPoolBalance = newStRwaBalance;
        constantK = newK;
        
        emit SwapStRWAToRWA(msg.sender, stRwaAmount, rwaAmount, fee);
        emit PoolUpdated(rwaPoolBalance, stRwaPoolBalance, constantK);
        
        return rwaAmount;
    }
    
    /**
     * @dev RWA → stRWA（使用 AMM 曲线）
     */
    function swapRWAToStRWA(uint256 rwaAmount) external nonReentrant returns (uint256) {
        require(rwaAmount > 0, "Amount must be greater than zero");
        require(rwaPoolBalance > 0 && stRwaPoolBalance > 0, "Pool not initialized");
        
        // 计算手续费
        uint256 fee = (rwaAmount * swapFeeRate) / 1000;
        uint256 rwaAmountAfterFee = rwaAmount - fee;
        
        // 计算输出金额（AMM 公式）
        // Δy = (y * Δx) / (x + Δx)
        uint256 stRwaAmount = (stRwaPoolBalance * rwaAmountAfterFee) / (rwaPoolBalance + rwaAmountAfterFee);
        
        // 检查最小输出
        require(stRwaAmount > 0, "Insufficient output");
        require(stRwaPoolBalance >= stRwaAmount, "Insufficient stRWA pool");
        
        // 检查池子健康度
        uint256 newRwaBalance = rwaPoolBalance + rwaAmountAfterFee;
        uint256 newStRwaBalance = stRwaPoolBalance - stRwaAmount;
        
        uint256 newK = newRwaBalance * newStRwaBalance;
        require(newK >= constantK * 99 / 100, "Pool imbalance detected");
        
        // 转账
        rwaToken.safeTransferFrom(msg.sender, address(this), rwaAmount);
        stRwaToken.safeTransfer(msg.sender, stRwaAmount);
        
        // 更新池子
        rwaPoolBalance = newRwaBalance;
        stRwaPoolBalance = newStRwaBalance;
        constantK = newK;
        
        emit SwapRWAToStRWA(msg.sender, rwaAmount, stRwaAmount, fee);
        emit PoolUpdated(rwaPoolBalance, stRwaPoolBalance, constantK);
        
        return stRwaAmount;
    }
    
    /**
     * @dev 计算预期输出金额（不执行交易）
     */
    function getSwapOutput(uint256 inputAmount, bool isStRWAToRWA) external view returns (uint256) {
        if (rwaPoolBalance == 0 || stRwaPoolBalance == 0) {
            return 0;
        }
        
        uint256 inputAmountAfterFee = inputAmount - (inputAmount * swapFeeRate / 1000);
        
        if (isStRWAToRWA) {
            return (rwaPoolBalance * inputAmountAfterFee) / (stRwaPoolBalance + inputAmountAfterFee);
        } else {
            return (stRwaPoolBalance * inputAmountAfterFee) / (rwaPoolBalance + inputAmountAfterFee);
        }
    }
    
    /**
     * @dev 设置互换手续费（仅 Owner）
     */
    function setSwapFeeRate(uint256 _swapFeeRate) external onlyOwner {
        require(_swapFeeRate <= 50, "Fee rate too high");  // 最高 5%
        swapFeeRate = _swapFeeRate;
    }
    
    /**
     * @dev 项目方补充流动性（紧急情况）
     */
    function addLiquidity(uint256 rwaAmount, uint256 stRwaAmount) external onlyOwner {
        if (rwaAmount > 0) {
            rwaToken.safeTransferFrom(msg.sender, address(this), rwaAmount);
            rwaPoolBalance += rwaAmount;
        }
        
        if (stRwaAmount > 0) {
            stRwaToken.safeTransferFrom(msg.sender, address(this), stRwaAmount);
            stRwaPoolBalance += stRwaAmount;
        }
        
        // 重新计算 k
        constantK = rwaPoolBalance * stRwaPoolBalance;
        
        emit PoolUpdated(rwaPoolBalance, stRwaPoolBalance, constantK);
    }
    
    /**
     * @dev 查询池子状态
     */
    function getPoolStatus() external view returns (
        uint256 rwaBalance,
        uint256 stRwaBalance,
        uint256 k,
        uint256 currentPrice  // 当前价格 = rwaBalance / stRwaBalance
    ) {
        uint256 price = 0;
        if (stRwaPoolBalance > 0) {
            price = (rwaPoolBalance * 1e18) / stRwaPoolBalance;  // 18位精度
        }
        
        return (rwaPoolBalance, stRwaPoolBalance, constantK, price);
    }
}
```

## 🚨 漏洞3: 烧伤机制深度优化

### 漏洞描述

**当前问题**：
- 单次奖励 ≤ 自身质押 × 50%
- 对于15代级差系统，1000块能拿500块，拨比依然快
- 没有每日总额限制

### 修复方案：每日动态收益总额限制

```solidity
// StakingContract.sol - 添加每日动态收益限制
struct UserInfo {
    // ... 现有字段 ...
    uint256 dailyDynamicRewardsReceived;  // 今日已领取的动态收益
    uint256 lastRewardDay;                // 上次领取收益的日期
}

uint256 public constant MAX_DAILY_DYNAMIC_REWARDS_RATIO = 15;  // 每日最多15%（可调整）

function updateUserRewards(
    address user,
    uint256 rwAmount,
    uint256 usdtAmount,
    uint256 rewardId
) external nonReentrant whenNotPaused {
    require(msg.sender == backendAddress, "Only backend can call");
    require(!processedRewards[rewardId], "Reward already processed");
    
    UserInfo storage userInfo = users[user];
    
    // ========== 新增：每日动态收益总额限制 ==========
    uint256 currentDay = block.timestamp / 1 days;
    
    // 如果是新的一天，重置计数器
    if (userInfo.lastRewardDay != currentDay) {
        userInfo.dailyDynamicRewardsReceived = 0;
        userInfo.lastRewardDay = currentDay;
    }
    
    // 检查每日限额：每日动态收益总额 ≤ 自身质押 × 15%
    uint256 maxDailyRewards = userInfo.totalStaked * MAX_DAILY_DYNAMIC_REWARDS_RATIO / 100;
    require(
        userInfo.dailyDynamicRewardsReceived + usdtAmount <= maxDailyRewards,
        "Daily dynamic rewards cap exceeded"
    );
    
    // ========== 现有：单次奖励限制（烧伤机制） ==========
    uint256 maxSingleReward = userInfo.totalStaked * MAX_SINGLE_REWARD_RATIO / 100;
    if (usdtAmount > maxSingleReward) {
        usdtAmount = maxSingleReward;
        emit RewardBurned(user, usdtAmount, block.timestamp);
    }
    
    // ========== 现有：个人累计封顶 ==========
    uint256 maxTotalRewards = userInfo.totalStaked * MAX_DYNAMIC_REWARDS_MULTIPLIER;
    require(
        userInfo.totalDynamicRewardsReceived + usdtAmount <= maxTotalRewards,
        "Personal reward cap exceeded, please restake"
    );
    
    // ========== 现有：软着陆机制 ==========
    uint256 globalRewardRatio = (totalDynamicRewardsPaid * 100) / totalStaked;
    if (globalRewardRatio >= SOFT_LANDING_THRESHOLD) {
        uint256 reductionRate = (globalRewardRatio - SOFT_LANDING_THRESHOLD) * 2;
        usdtAmount = usdtAmount * (100 - reductionRate) / 100;
    }
    
    // ========== 现有：50% 全局上限检查 ==========
    require(
        totalDynamicRewardsPaid + usdtAmount <= totalStaked * 50 / 100,
        "Dynamic rewards exceed 50% limit"
    );
    
    // ========== 状态更新 ==========
    processedRewards[rewardId] = true;
    userInfo.rwaPending += rwAmount;
    userInfo.usdtRewards += usdtAmount;
    userInfo.totalDynamicRewardsReceived += usdtAmount;
    userInfo.dailyDynamicRewardsReceived += usdtAmount;  // 新增
    totalDynamicRewardsPaid += usdtAmount;
    
    emit RewardsUpdated(user, rwAmount, usdtAmount, rewardId, block.timestamp);
}

/**
 * @dev 查询用户每日剩余动态收益额度
 */
function getDailyRemainingRewardCapacity(address user) external view returns (uint256) {
    UserInfo storage userInfo = users[user];
    
    uint256 currentDay = block.timestamp / 1 days;
    uint256 dailyReceived = (userInfo.lastRewardDay == currentDay) 
        ? userInfo.dailyDynamicRewardsReceived 
        : 0;
    
    uint256 maxDailyRewards = userInfo.totalStaked * MAX_DAILY_DYNAMIC_REWARDS_RATIO / 100;
    
    return (maxDailyRewards > dailyReceived) ? maxDailyRewards - dailyReceived : 0;
}
```

```typescript
        const ancestorInfo = await this.getUserInfo(ancestor.ancestor_address);
        
        // 检查今日已领取的动态收益
        const currentDay = Math.floor(Date.now() / (24 * 60 * 60 * 1000));
        const lastRewardDay = ancestorInfo.last_reward_day || 0;
        const dailyReceived = (lastRewardDay === currentDay) 
            ? new BigNumber(ancestorInfo.daily_dynamic_rewards_received || '0')
            : new BigNumber(0);
        
        // 计算每日限额：自身质押 × 15%
        const maxDailyRewards = new BigNumber(ancestorInfo.total_staked)
            .multipliedBy(0.15);
        
        // 计算本次奖励
        const calculatedReward = stakeAmountBN
            .multipliedBy(differentialPercentage)
            .integerValue(BigNumber.ROUND_DOWN);
        
        // 烧伤机制：单次奖励 ≤ 自身质押 × 50%
        const ancestorStaked = new BigNumber(ancestorInfo.total_staked);
        const maxSingleReward = ancestorStaked.multipliedBy(0.5);
        let actualReward = BigNumber.min(calculatedReward, maxSingleReward);
        
        // ========== 新增：每日限额检查 ==========
        const remainingDailyCapacity = maxDailyRewards.minus(dailyReceived);
        if (actualReward.isGreaterThan(remainingDailyCapacity)) {
            actualReward = remainingDailyCapacity;
            if (actualReward.isLessThanOrEqualTo(0)) {
                continue;  // 今日额度已用完，跳过
            }
        }
        
        if (actualReward.gt(0)) {
            rewards.push({
                beneficiary: ancestor.ancestor_address,
                amount: actualReward.toString(),
                percentage: differentialPercentage,
                nodeLevel: ancestorLevel,
                fromUser: userAddress,
                stakeId: parseInt(stakeId),
                burned: calculatedReward.isGreaterThan(actualReward)
            });
        }
    }
    
    return rewards;
}
```

**推荐参数**：
- 每日限额：15%（可根据实际情况调整 10%-20%）
- 单次限额：50%（保持不变）
- 累计限额：3倍（保持不变）

**优势**：
- ✅ 强制大团长不断复投，锁定资金
- ✅ 防止单日大量提取，平滑资金流出
- ✅ 降低拨比，提高可持续性

### 4.2 stRWA 铸造限额

```solidity
// StRWA.sol - 添加铸造限额
contract StRWA is ERC20, Ownable {
    address public stakingContract;
    
    // 铸造限额
    uint256 public maxMintPerCall = 1000000 * 10**18;  // 单次最大铸造量
    uint256 public dailyMintLimit = 10000000 * 10**18; // 每日最大铸造量
    uint256 public dailyMintUsed;                      // 每日已使用额度
    uint256 public lastMintDay;                        // 上次铸造日期
    
    /**
     * @dev 铸造 stRWA（仅质押合约可调用）
     */
    function mint(address to, uint256 amount) external {
        require(msg.sender == stakingContract, "Only staking contract");
        require(amount > 0, "Amount must be greater than zero");
        require(amount <= maxMintPerCall, "Exceeds max mint per call");
        
        // 检查每日限额
        uint256 currentDay = block.timestamp / 1 days;
        if (lastMintDay != currentDay) {
            dailyMintUsed = 0;
            lastMintDay = currentDay;
        }
        
        require(
            dailyMintUsed + amount <= dailyMintLimit,
            "Daily mint limit exceeded"
        );
        
        dailyMintUsed += amount;
        _mint(to, amount);
        
        emit Minted(to, amount, block.timestamp);
    }
    
    /**
     * @dev 设置铸造限额（仅 Owner）
     */
    function setMintLimits(
        uint256 _maxMintPerCall,
        uint256 _dailyMintLimit
    ) external onlyOwner {
        require(_maxMintPerCall > 0, "Invalid max mint per call");
        require(_dailyMintLimit > 0, "Invalid daily mint limit");
        
        maxMintPerCall = _maxMintPerCall;
        dailyMintLimit = _dailyMintLimit;
        
        emit MintLimitsUpdated(_maxMintPerCall, _dailyMintLimit);
    }
    
    event Minted(address indexed to, uint256 amount, uint256 timestamp);
    event MintLimitsUpdated(uint256 maxMintPerCall, uint256 dailyMintLimit);
}
```
